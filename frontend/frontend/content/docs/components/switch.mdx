# Switch Component

Toggle control for binary on/off states.

## Basic Usage

```tsx
import { Switch } from '@/components/ui/switch'

function Example() {
  return <Switch />
}
```

## Controlled Switch

```tsx
function ControlledSwitch() {
  const [enabled, setEnabled] = useState(false)

  return (
    <Switch checked={enabled} onCheckedChange={setEnabled} />
  )
}
```

## Props

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `checked` | `boolean` | - | Controlled checked state |
| `defaultChecked` | `boolean` | `false` | Default checked state |
| `onCheckedChange` | `(checked: boolean) => void` | - | Change callback |
| `disabled` | `boolean` | `false` | Disable switch |
| `required` | `boolean` | `false` | Required field |
| `name` | `string` | - | Form field name |

### Confirmation Dialog

```tsx
function SwitchWithConfirmation() {
  const [enabled, setEnabled] = useState(false)
  const [showDialog, setShowDialog] = useState(false)
  const [pendingValue, setPendingValue] = useState(false)

  const handleChange = (checked: boolean) => {
    if (checked) {
      setPendingValue(checked)
      setShowDialog(true)
    } else {
      setEnabled(checked)
    }
  }

  const handleConfirm = () => {
    setEnabled(pendingValue)
    setShowDialog(false)
  }

  return (
    <>
      <div className="flex items-center space-x-2">
        <Switch checked={enabled} onCheckedChange={handleChange} />
        <Label>Enable dangerous feature</Label>
      </div>

      <Dialog open={showDialog} onOpenChange={setShowDialog}>
        <DialogContent>
          <DialogHeader>
            <DialogTitle>Are you sure?</DialogTitle>
            <DialogDescription>
              This will enable a potentially dangerous feature.
            </DialogDescription>
          </DialogHeader>
          <DialogFooter>
            <Button variant="outline" onClick={() => setShowDialog(false)}>
              Cancel
            </Button>
            <Button onClick={handleConfirm}>Confirm</Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>
    </>
  )
}
```

### Group of Switches

## Best Practices

1. **Clear labels** - Always provide descriptive labels
2. **Immediate feedback** - Changes should take effect immediately
3. **Confirmation for critical actions** - Use dialogs for dangerous operations
4. **Loading states** - Show loading for async operations
5. **Accessible** - Use proper ARIA attributes and keyboard support
6. **Consistent placement** - Keep switches in predictable locations

## Testing

```typescript
import { render, screen, fireEvent } from '@testing-library/react'
import { Switch } from './switch'

describe('Switch', () => {
  it('toggles on click', () => {
    const handleChange = jest.fn()
    render(<Switch onCheckedChange={handleChange} />)

    const switchElement = screen.getByRole('switch')
    fireEvent.click(switchElement)

    expect(handleChange).toHaveBeenCalledWith(true)
  })

  it('can be disabled', () => {
    render(<Switch disabled />)
    
    const switchElement = screen.getByRole('switch')
    expect(switchElement).toBeDisabled()
  })
})
```
