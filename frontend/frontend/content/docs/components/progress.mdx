# Progress Component

Display progress indicators for tasks, uploads, and loading states.

## Basic Usage

```tsx
import { Progress } from '@/components/ui/progress'

function Example() {
  return <Progress value={60} />
}
```

## Variants

```tsx
// Default
<Progress value={60} />

// With color
<Progress value={60} className="[&>div]:bg-green-500" />

// Indeterminate
<Progress />
```

## Props

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `value` | `number` | - | Progress value (0-100) |
| `max` | `number` | `100` | Maximum value |
| `className` | `string` | - | Additional CSS classes |

## Examples

### With Label

### File Upload Progress

```tsx
function FileUploadProgress() {
  const [progress, setProgress] = useState(0)
  const [isUploading, setIsUploading] = useState(false)

  const handleUpload = async (file: File) => {
    setIsUploading(true)
    setProgress(0)

    // Simulate upload progress
    const interval = setInterval(() => {
      setProgress((prev) => {
        if (prev >= 100) {
          clearInterval(interval)
          setIsUploading(false)
          return 100
        }
        return prev + 10
      })
    }, 500)
  }

  return (
    <div className="space-y-4">
      <Input type="file" onChange={(e) => handleUpload(e.target.files?.[0]!)} />
      {isUploading && (
        <div className="space-y-2">
          <div className="flex justify-between text-sm">
            <span>Uploading...</span>
            <span>{progress}%</span>
          </div>
          <Progress value={progress} />
        </div>
      )}
    </div>
  )
}
```

### Multi-Step Progress

```tsx
function MultiStepProgress() {
  const [currentStep, setCurrentStep] = useState(1)
  const totalSteps = 4
  const progress = (currentStep / totalSteps) * 100

  return (
    <div className="space-y-4">
      <div className="flex justify-between text-sm">
        <span>Step {currentStep} of {totalSteps}</span>
        <span>{Math.round(progress)}%</span>
      </div>
      <Progress value={progress} />
      <div className="flex gap-2">
        <Button
          onClick={() => setCurrentStep(Math.max(1, currentStep - 1))}
          disabled={currentStep === 1}
        >
          Previous
        </Button>
        <Button
          onClick={() => setCurrentStep(Math.min(totalSteps, currentStep + 1))}
          disabled={currentStep === totalSteps}
        >
          Next
        </Button>
      </div>
    </div>
  )
}
```

### Circular Progress

```tsx
function CircularProgress({ value }: { value: number }) {
  const radius = 40
  const circumference = 2 * Math.PI * radius
  const offset = circumference - (value / 100) * circumference

  return (
    <div className="relative inline-flex items-center justify-center">
      <svg className="transform -rotate-90" width="100" height="100">
        <circle
          cx="50"
          cy="50"
          r={radius}
          stroke="currentColor"
          strokeWidth="8"
          fill="none"
          className="text-secondary"
        />
        <circle
          cx="50"
          cy="50"
          r={radius}
          stroke="currentColor"
          strokeWidth="8"
          fill="none"
          strokeDasharray={circumference}
          strokeDashoffset={offset}
          className="text-primary transition-all duration-300"
        />
      </svg>
      <span className="absolute text-xl font-semibold">{value}%</span>
    </div>
  )
}
```

### Progress with Status

```tsx
function ProgressWithStatus() {
  const [progress, setProgress] = useState(0)

  const getStatus = () => {
    if (progress < 30) return { text: 'Starting...', color: 'text-blue-500' }
    if (progress < 70) return { text: 'In Progress...', color: 'text-yellow-500' }
    if (progress < 100) return { text: 'Almost Done...', color: 'text-orange-500' }
    return { text: 'Complete!', color: 'text-green-500' }
  }

  const status = getStatus()

  return (
    <div className="space-y-2">
      <div className="flex justify-between text-sm">
        <span className={status.color}>{status.text}</span>
        <span>{progress}%</span>
      </div>
      <Progress value={progress} />
    </div>
  )
}
```

### Segmented Progress

```tsx
function SegmentedProgress({ segments, current }: { segments: number; current: number }) {
```

### Progress with Time Remaining

```tsx
function ProgressWithTime() {
  const [progress, setProgress] = useState(0)
  const [timeRemaining, setTimeRemaining] = useState(60)

  useEffect(() => {
    const interval = setInterval(() => {
      setProgress((prev) => Math.min(prev + 1, 100))
      setTimeRemaining((prev) => Math.max(prev - 0.6, 0))
    }, 600)

    return () => clearInterval(interval)
  }, [])

  const formatTime = (seconds: number) => {
    const mins = Math.floor(seconds / 60)
    const secs = Math.floor(seconds % 60)
    return `${mins}:${secs.toString().padStart(2, '0')}`
  }

  return (
    <div className="space-y-2">
      <div className="flex justify-between text-sm">
        <span>Processing...</span>
        <span>{formatTime(timeRemaining)} remaining</span>
      </div>
      <Progress value={progress} />
    </div>
  )
}
```

### Stacked Progress

```tsx
function StackedProgress({ items }: { items: Array<{ label: string; value: number; color: string }> }) {

/>
```

### Progress with Milestones

```tsx
function ProgressWithMilestones() {
  const [progress, setProgress] = useState(35)
  const milestones = [0, 25, 50, 75, 100]

  return (
    <div className="space-y-2">
      <Progress value={progress} />
      <div className="relative">
        <div className="flex justify-between">
          {milestones.map((milestone) => (
            <div
              key={milestone}
              className={`flex flex-col items-center ${
                progress >= milestone ? 'text-primary' : 'text-muted-foreground'
              }`}
            >
              <div
                className={`h-3 w-3 rounded-full border-2 ${
                  progress >= milestone
                    ? 'bg-primary border-primary'
                    : 'bg-background border-muted-foreground'
                }`}
              />
              <span className="text-xs mt-1">{milestone}%</span>
            </div>
          ))}
        </div>
      </div>
    </div>
  )
}
```

## Best Practices

1. **Show percentage** - Display numeric progress when possible
2. **Provide context** - Explain what's being loaded/processed
3. **Estimate time** - Show time remaining for long operations
4. **Use appropriate type** - Determinate for known progress, indeterminate for unknown
5. **Keep visible** - Don't hide progress indicators
6. **Smooth animation** - Use transitions for visual appeal
