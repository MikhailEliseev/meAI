# Textarea Component

Multi-line text input field for longer form content.

## Basic Usage

```tsx
import { Textarea } from '@/components/ui/textarea'

function Example() {
  return <Textarea placeholder="Type your message here." />
}
```

## Props

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `placeholder` | `string` | - | Placeholder text |
| `value` | `string` | - | Controlled value |
| `defaultValue` | `string` | - | Default value |
| `onChange` | `(e: ChangeEvent) => void` | - | Change handler |
| `disabled` | `boolean` | `false` | Disable textarea |
| `readOnly` | `boolean` | `false` | Read-only mode |
| `rows` | `number` | `3` | Number of visible rows |
| `maxLength` | `number` | - | Maximum character count |
| `required` | `boolean` | `false` | Required field |
| `className` | `string` | - | Additional CSS classes |

### Auto-resize Textarea

```tsx
function AutoResizeTextarea() {
  const textareaRef = useRef<HTMLTextAreaElement>(null)

  const handleInput = () => {
    const textarea = textareaRef.current
    if (textarea) {
      textarea.style.height = 'auto'
      textarea.style.height = `${textarea.scrollHeight}px`
    }
  }

  return (
    <Textarea
      ref={textareaRef}
      onInput={handleInput}
      placeholder="Type something..."
      rows={1}
    />
  )
}
```

### With Validation

```tsx
function ValidatedTextarea() {
  const [value, setValue] = useState('')
  const [error, setError] = useState('')

  const validate = (text: string) => {
    if (text.length < 10) {
      setError('Message must be at least 10 characters')
    } else {
      setError('')
    }
  }

  return (
    <div className="space-y-2">
      <Textarea
        value={value}
        onChange={(e) => {
          setValue(e.target.value)
          validate(e.target.value)
        }}
        placeholder="Enter your message"
        className={error ? 'border-destructive' : ''}
      />
      {error && (
        <p className="text-sm text-destructive">{error}</p>
      )}
    </div>
  )
}
```

### Mention Support

```tsx
function TextareaWithMentions() {
  const [value, setValue] = useState('')
  const [showSuggestions, setShowSuggestions] = useState(false)
  const [mentionQuery, setMentionQuery] = useState('')

  const users = ['alice', 'bob', 'charlie']

  const handleChange = (e: React.ChangeEvent<HTMLTextAreaElement>) => {
    const text = e.target.value
    setValue(text)

    // Check for @ mention
    const lastWord = text.split(/\s/).pop() || ''
    if (lastWord.startsWith('@')) {
      setMentionQuery(lastWord.slice(1))
      setShowSuggestions(true)
    } else {
      setShowSuggestions(false)
    }
  }

  const insertMention = (user: string) => {
    const words = value.split(/\s/)
    words[words.length - 1] = `@${user} `
    setValue(words.join(' '))
    setShowSuggestions(false)
  }

  const filteredUsers = users.filter((user) =>
    user.toLowerCase().includes(mentionQuery.toLowerCase())
  )

  return (
    <div className="relative space-y-2">
      <Textarea
        value={value}
        onChange={handleChange}
        placeholder="Type @ to mention someone..."
      />
      {showSuggestions && filteredUsers.length > 0 && (
        <div className="absolute z-10 w-full border rounded-md bg-popover p-1 shadow-md">
          {filteredUsers.map((user) => (
            <button
              key={user}
              onClick={() => insertMention(user)}
              className="w-full text-left px-2 py-1 hover:bg-accent rounded-sm"
            >
              @{user}
            </button>
          ))}
        </div>
      )}
    </div>
  )
}
```

## Best Practices

1. **Clear labels** - Always provide descriptive labels
2. **Placeholder text** - Use helpful placeholder text
3. **Character limits** - Show character/word count for limited fields
4. **Validation** - Provide clear error messages
5. **Auto-resize** - Consider auto-resize for better UX
6. **Accessibility** - Use proper ARIA attributes

## Testing

```typescript
import { render, screen, fireEvent } from '@testing-library/react'
import { Textarea } from './textarea'

describe('Textarea', () => {
  it('handles input', () => {
    const handleChange = jest.fn()
    render(<Textarea onChange={handleChange} />)

    const textarea = screen.getByRole('textbox')
    fireEvent.change(textarea, { target: { value: 'Hello' } })

    expect(handleChange).toHaveBeenCalled()
    expect(textarea).toHaveValue('Hello')
  })

  it('respects maxLength', () => {
    render(<Textarea maxLength={10} />)
    
    const textarea = screen.getByRole('textbox')
    expect(textarea).toHaveAttribute('maxLength', '10')
  })
})
```
