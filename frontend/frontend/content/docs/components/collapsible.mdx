# Collapsible Component

Expandable and collapsible content sections for showing and hiding content.

## Basic Usage

```tsx
import {
  Collapsible,
  CollapsibleContent,
  CollapsibleTrigger,
} from '@/components/ui/collapsible'

function Example() {
  const [isOpen, setIsOpen] = useState(false)

  return (
    <Collapsible open={isOpen} onOpenChange={setIsOpen}>
      <CollapsibleTrigger asChild>
        <Button variant="ghost">
          Toggle
          <ChevronDown className={`ml-2 h-4 w-4 transition-transform ${isOpen ? 'rotate-180' : ''}`} />
        </Button>
      </CollapsibleTrigger>
      <CollapsibleContent>
        <div className="p-4 border rounded-lg mt-2">
          Collapsible content goes here
        </div>
      </CollapsibleContent>
    </Collapsible>
  )
}
```

## Props

### Collapsible

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `open` | `boolean` | - | Controlled open state |
| `defaultOpen` | `boolean` | `false` | Default open state |
| `onOpenChange` | `(open: boolean) => void` | - | Open state callback |
| `disabled` | `boolean` | `false` | Disable collapsible |

function FAQCollapsible() {

### Sidebar Menu

```tsx
function SidebarMenu() {
  const [openItems, setOpenItems] = useState<string[]>([])

  const menuItems = [
    {
      title: 'Dashboard',
      icon: Home,
      items: ['Overview', 'Analytics', 'Reports'],
    },
    {
      title: 'Products',
      icon: Package,
      items: ['All Products', 'Categories', 'Inventory'],
    },
    {
      title: 'Settings',
      icon: Settings,
      items: ['Profile', 'Preferences', 'Security'],
    },
  ]

  const toggleItem = (title: string) => {
    setOpenItems((prev) =>
      prev.includes(title)
        ? prev.filter((item) => item !== title)
        : [...prev, title]
    )
  }

  return (
    <div className="w-64 space-y-2">
      {menuItems.map((item) => (
        <Collapsible
          key={item.title}
          open={openItems.includes(item.title)}
          onOpenChange={() => toggleItem(item.title)}
        >
          <CollapsibleTrigger asChild>
            <Button
              variant="ghost"
              className="w-full justify-between"
            >
              <div className="flex items-center gap-2">
                <item.icon className="h-4 w-4" />
                <span>{item.title}</span>
              </div>
              <ChevronDown
                className={`h-4 w-4 transition-transform ${
                  openItems.includes(item.title) ? 'rotate-180' : ''
                }`}
              />
            </Button>
          </CollapsibleTrigger>
          <CollapsibleContent>
            <div className="ml-6 space-y-1 py-2">
              {item.items.map((subItem) => (
                <Button
                  key={subItem}
                  variant="ghost"
                  className="w-full justify-start"
                  size="sm"
                >
                  {subItem}
                </Button>
              ))}
            </div>
          </CollapsibleContent>
        </Collapsible>
      ))}
    </div>
  )
}
```

### Table Row Expansion

```tsx
function ExpandableTable() {
  const [expandedRows, setExpandedRows] = useState<number[]>([])

  const data = [
    { id: 1, name: 'Product 1', price: '$100', details: 'Detailed info 1' },
    { id: 2, name: 'Product 2', price: '$200', details: 'Detailed info 2' },
  ]

  const toggleRow = (id: number) => {
    setExpandedRows((prev) =>
      prev.includes(id) ? prev.filter((rowId) => rowId !== id) : [...prev, id]
    )
  }

  return (
    <Table>
      <TableHeader>
        <TableRow>
          <TableHead className="w-12"></TableHead>
          <TableHead>Name</TableHead>
          <TableHead>Price</TableHead>
        </TableRow>
      </TableHeader>
      <TableBody>
        {data.map((row) => (
          <React.Fragment key={row.id}>
            <TableRow>
              <TableCell>
                <Button
                  variant="ghost"
                  size="icon"
                  onClick={() => toggleRow(row.id)}
                >
                  <ChevronDown
                    className={`h-4 w-4 transition-transform ${
                      expandedRows.includes(row.id) ? 'rotate-180' : ''
                    }`}
                  />
                </Button>
              </TableCell>
              <TableCell>{row.name}</TableCell>
              <TableCell>{row.price}</TableCell>
            </TableRow>
            {expandedRows.includes(row.id) && (
              <TableRow>
                <TableCell colSpan={3} className="bg-muted/50">
                  <div className="p-4">
                    <p className="text-sm">{row.details}</p>
                  </div>
                </TableCell>
              </TableRow>
            )}
          </React.Fragment>
        ))}
      </TableBody>
    </Table>
  )
}
```

### Animated Collapsible

```tsx
function AnimatedCollapsible() {
  const [isOpen, setIsOpen] = useState(false)

  return (
    <Collapsible open={isOpen} onOpenChange={setIsOpen}>
      <CollapsibleTrigger asChild>
        <Button variant="outline" className="w-full justify-between">
          <span>Toggle Animation</span>
          <motion.div
            animate={{ rotate: isOpen ? 180 : 0 }}
            transition={{ duration: 0.2 }}
          >
            <ChevronDown className="h-4 w-4" />
          </motion.div>
        </Button>
      </CollapsibleTrigger>
      <CollapsibleContent>
        <motion.div
          initial={{ opacity: 0, y: -10 }}
          animate={{ opacity: 1, y: 0 }}
          exit={{ opacity: 0, y: -10 }}
          transition={{ duration: 0.2 }}
          className="p-4 border rounded-lg mt-2"
        >
          Animated content
        </motion.div>
      </CollapsibleContent>
    </Collapsible>
  )
}
```

## Best Practices

1. **Clear indicators** - Show expand/collapse state clearly
2. **Smooth transitions** - Use animations for better UX
3. **Keyboard support** - Allow keyboard navigation
4. **Preserve state** - Remember open/closed state when appropriate
5. **Accessible** - Use proper ARIA attributes
6. **Performance** - Lazy load content when possible

## Accessibility

```tsx
<Collapsible>
  <CollapsibleTrigger
    aria-expanded={isOpen}
    aria-controls="collapsible-content"
  >
    Toggle
  </CollapsibleTrigger>
  <CollapsibleContent
    id="collapsible-content"
    role="region"
  >
    Content
  </CollapsibleContent>
</Collapsible>
```

## Keyboard Navigation

- `Space/Enter` - Toggle collapsible
- `Tab` - Navigate to next focusable element

## Testing

```typescript
import { render, screen, fireEvent } from '@testing-library/react'
import { Collapsible, CollapsibleTrigger, CollapsibleContent } from './collapsible'

describe('Collapsible', () => {
  it('toggles content', () => {
    render(
      <Collapsible>
        <CollapsibleTrigger>Toggle</CollapsibleTrigger>
        <CollapsibleContent>Content</CollapsibleContent>
      </Collapsible>
    )

    expect(screen.queryByText('Content')).not.toBeVisible()

    fireEvent.click(screen.getByText('Toggle'))
    expect(screen.getByText('Content')).toBeVisible()
  })

  it('respects defaultOpen', () => {
    render(
      <Collapsible defaultOpen>
        <CollapsibleTrigger>Toggle</CollapsibleTrigger>
        <CollapsibleContent>Content</CollapsibleContent>
      </Collapsible>
    )

    expect(screen.getByText('Content')).toBeVisible()
  })
})
```
