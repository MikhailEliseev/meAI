# Slider Component

Input slider for selecting numeric values from a range.

## Basic Usage

```tsx
import { Slider } from '@/components/ui/slider'

function Example() {
  const [value, setValue] = useState([50])

  return (
    <Slider
      value={value}
      onValueChange={setValue}
      max={100}
      step={1}
    />
  )
}
```

## Props

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `value` | `number[]` | - | Controlled value |
| `defaultValue` | `number[]` | `[0]` | Default value |
| `onValueChange` | `(value: number[]) => void` | - | Change callback |
| `min` | `number` | `0` | Minimum value |
| `max` | `number` | `100` | Maximum value |
| `step` | `number` | `1` | Step increment |
| `disabled` | `boolean` | `false` | Disable slider |
| `orientation` | `'horizontal' \| 'vertical'` | `'horizontal'` | Slider orientation |

### Volume Control

```tsx
function VolumeControl() {
  const [value, setValue] = useState([70])
  const [muted, setMuted] = useState(false)

  return (
    <div className="space-y-4">
      <div className="flex items-center gap-4">
        <Button
          variant="ghost"
          size="icon"
          onClick={() => setMuted(!muted)}
        >
          {muted || value[0] === 0 ? (
            <VolumeX className="h-4 w-4" />
          ) : value[0] < 50 ? (
            <Volume1 className="h-4 w-4" />
          ) : (
            <Volume2 className="h-4 w-4" />
          )}
        </Button>
        <Slider
          value={muted ? [0] : value}
          onValueChange={(v) => {
            setValue(v)
            if (v[0] > 0) setMuted(false)
          }}
          max={100}
          step={1}
          className="flex-1"
        />
        <span className="text-sm text-muted-foreground w-12 text-right">
          {muted ? 0 : value[0]}%
        </span>
      </div>
    </div>
  )
}
```

### Time Range Selector

```tsx
function TimeRangeSelector() {
  const [value, setValue] = useState([9, 17])

  const formatTime = (hour: number) => {
    const period = hour >= 12 ? 'PM' : 'AM'
    const displayHour = hour > 12 ? hour - 12 : hour === 0 ? 12 : hour
    return `${displayHour}:00 ${period}`
  }

  return (
    <div className="space-y-4">
      <div className="flex items-center justify-between">
        <Label>Working Hours</Label>
        <span className="text-sm text-muted-foreground">
          {formatTime(value[0])} - {formatTime(value[1])}
        </span>
      </div>
      <Slider
        value={value}
        onValueChange={setValue}
        min={0}
        max={24}
        step={1}
      />
    </div>
  )
}
```

## Best Practices

1. **Show value** - Display current value for clarity
2. **Use steps** - Set appropriate step increments
3. **Add labels** - Label min/max or key values
4. **Provide feedback** - Show value changes clearly
5. **Consider range** - Use range slider for min/max selection
6. **Keyboard support** - Ensure arrow key navigation works

## Accessibility

```tsx

  step={1}
  aria-label="Volume control"
  aria-valuemin={0}
  aria-valuemax={100}
  aria-valuenow={value[0]}

```

## Keyboard Navigation

- `Arrow Left/Down` - Decrease value
- `Arrow Right/Up` - Increase value
- `Home` - Set to minimum
- `End` - Set to maximum
- `Page Up/Down` - Large increment/decrement

## Testing

```typescript
import { render, screen, fireEvent } from '@testing-library/react'
import { Slider } from './slider'

describe('Slider', () => {
  it('changes value on interaction', () => {
    const handleChange = jest.fn()
    render(
      <Slider
        value={[50]}
        onValueChange={handleChange}
        max={100}
      />
    )

    const slider = screen.getByRole('slider')
    fireEvent.keyDown(slider, { key: 'ArrowRight' })

    expect(handleChange).toHaveBeenCalled()
  })

  it('respects min and max', () => {
    const handleChange = jest.fn()
    render(
      <Slider
        value={[50]}
        onValueChange={handleChange}
        min={0}
        max={100}
      />
    )

    const slider = screen.getByRole('slider')
    fireEvent.keyDown(slider, { key: 'Home' })

    expect(handleChange).toHaveBeenCalledWith([0])
  })
})
```
