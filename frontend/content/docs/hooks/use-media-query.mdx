# useMediaQuery Hook

React hook for responsive design using CSS media queries.

## Basic Usage

```tsx
import { useMediaQuery } from '@/hooks/useMediaQuery'

function ResponsiveComponent() {
  const isMobile = useMediaQuery('(max-width: 768px)')

  return (
    <div>
      {isMobile ? (
        <MobileLayout />
      ) : (
        <DesktopLayout />
      )}
    </div>
  )
}
```

## Parameters

| Parameter | Type | Required | Description |
|-----------|------|----------|-------------|
| `query` | `string` | Yes | CSS media query string |

## Return Value

Returns `boolean` - `true` if media query matches, `false` otherwise.

## Common Breakpoints

```tsx
function useBreakpoints() {
  const isMobile = useMediaQuery('(max-width: 640px)')
  const isTablet = useMediaQuery('(min-width: 641px) and (max-width: 1024px)')
  const isDesktop = useMediaQuery('(min-width: 1025px)')
  const isLargeDesktop = useMediaQuery('(min-width: 1440px)')

  return { isMobile, isTablet, isDesktop, isLargeDesktop }
}
```

function ProductGrid({ products }: { products: Product[] }) {

## Implementation

```typescript
import { useState, useEffect } from 'react'

export function useMediaQuery(query: string): boolean {
  const [matches, setMatches] = useState(false)

  useEffect(() => {
    // Check if window is defined (SSR)
    if (typeof window === 'undefined') {
      return
    }

    const mediaQuery = window.matchMedia(query)
    
    // Set initial value
    setMatches(mediaQuery.matches)

    // Create event listener
    const handleChange = (event: MediaQueryListEvent) => {
      setMatches(event.matches)
    }

    // Add listener
    mediaQuery.addEventListener('change', handleChange)

    // Cleanup
    return () => {
      mediaQuery.removeEventListener('change', handleChange)
    }
  }, [query])

  return matches
}
```

## SSR-Safe Implementation

```typescript
export function useMediaQuery(query: string): boolean {
  const [matches, setMatches] = useState(false)
  const [mounted, setMounted] = useState(false)

  useEffect(() => {
    setMounted(true)
  }, [])

  useEffect(() => {
    if (!mounted) return

    const mediaQuery = window.matchMedia(query)
    setMatches(mediaQuery.matches)

    const handleChange = (event: MediaQueryListEvent) => {
      setMatches(event.matches)
    }

    mediaQuery.addEventListener('change', handleChange)
    return () => mediaQuery.removeEventListener('change', handleChange)
  }, [query, mounted])

  return matches
}
```

const breakpoints = {

## Multiple Queries

```typescript
export function useMediaQueries(queries: Record<string, string>) {
  const [matches, setMatches] = useState<Record<string, boolean>>({})

  useEffect(() => {
    const mediaQueries = Object.entries(queries).map(([key, query]) => ({
      key,
      query: window.matchMedia(query),
    }))

    const updateMatches = () => {
      setMatches(
        mediaQueries.reduce((acc, { key, query }) => ({
          ...acc,
          [key]: query.matches,
        }), {})
      )
    }

    updateMatches()

    mediaQueries.forEach(({ query }) => {
      query.addEventListener('change', updateMatches)
    })

    return () => {
      mediaQueries.forEach(({ query }) => {
        query.removeEventListener('change', updateMatches)
      })
    }
  }, [queries])

  return matches
}

// Usage
function Component() {
  const { isMobile, isTablet, isDesktop } = useMediaQueries({
    isMobile: '(max-width: 640px)',
    isTablet: '(min-width: 641px) and (max-width: 1024px)',
    isDesktop: '(min-width: 1025px)',
  })
}
```

## Common Media Queries

```typescript
// Screen sizes
'(max-width: 640px)'  // Mobile
'(min-width: 768px)'  // Tablet and up
'(min-width: 1024px)' // Desktop and up

// Orientation
'(orientation: portrait)'
'(orientation: landscape)'

// Color scheme
'(prefers-color-scheme: dark)'
'(prefers-color-scheme: light)'

// Motion
'(prefers-reduced-motion: reduce)'
'(prefers-reduced-motion: no-preference)'

// Hover capability
'(hover: hover)'      // Can hover (mouse)
'(hover: none)'       // Cannot hover (touch)

// Pointer precision
'(pointer: fine)'     // Precise pointer (mouse)
'(pointer: coarse)'   // Imprecise pointer (touch)

// Display mode
'(display-mode: standalone)' // PWA
'(display-mode: browser)'    // Browser

// Print
'print'
```

## Best Practices

1. **Use semantic breakpoints** - Name based on content, not devices
2. **Mobile-first** - Start with mobile layout, enhance for larger screens
3. **Avoid too many breakpoints** - 3-4 breakpoints usually sufficient
4. **Test on real devices** - Emulators don't always match reality
5. **Consider touch** - Use hover/pointer queries for touch detection
6. **SSR-safe** - Handle server-side rendering properly
7. **Performance** - Minimize re-renders with proper dependencies
