    ```python
    logger = logging.getLogger(__name__)

    TEMPLATES_DIR = Path(__file__).parent.parent / "templates"

    def generate_report_html(
        clinic_name: str,
        sections: dict,
        output_path: Path
    ) -> bool:
        """Generate HTML report using canonical template.

        Args:
            clinic_name: Clinic name for title
            sections: Dict with keys matching template placeholders:
                {
                    "clinic_overview_html": "<p>...</p>",
                    "competitors_html": "<div>...</div>",

                    "offer_html": "..."

            output_path: Where to save the report

            True if successful, False otherwise
        """
        try:
            # Load template
            env = Environment(loader=FileSystemLoader(TEMPLATES_DIR))
            template = env.get_template("report-template.html")

            # Render with sections
            html = template.render(
                clinic_name=clinic_name,
                **sections
            )

            # Write to file
            output_path.write_text(html, encoding="utf-8")
            logger.info("Report generated: %s (%d bytes)", output_path, len(html))
            return True

            logger.exception("Failed to generate report")
            return False
    ```

    ```python
    """
    generate_report — Hermes tool: Generate & Publish Presale Report

    Generates HTML report using canonical template, publishes to WordPress,
    returns report URL.

    Per D-16: WordPress REST API publishing via POST /wp-json/wp/v2/pages
    Per D-19: Returns report URL for SSE event
    """

    import json
    import logging
    from pathlib import Path
    import httpx
    from tools.registry import registry
    from scripts.generate_html_report import generate_report_html, sections_from_orchestrator_output

    logger = logging.getLogger(__name__)

    WP_API_BASE = "http://wordpress:80"  # Internal Docker network
    WP_AUTH = None  # Will use application password or JWT


    async def handle_generate_report(
        session_hash: str,
        clinic_name: str,
        orchestrator_data: dict,
        **kwargs
    ) -> str:
        """Generate HTML report and publish to WordPress.

        Args:
            session_hash: 8-char session identifier
            clinic_name: Clinic name for title and slug
            orchestrator_data: Full output from 3-pass orchestrator

        Returns:
            JSON with report URL and status
        """
        try:
            # 1. Generate HTML from template
            sections = sections_from_orchestrator_output(orchestrator_data)
            output_path = Path(f"/opt/data/memories/proposals/{session_hash}/report.html")
            output_path.parent.mkdir(parents=True, exist_ok=True)

            success = generate_report_html(clinic_name, sections, output_path)
            if not success:
                return json.dumps({"error": "Failed to generate report HTML"})

            # 2. Read generated HTML
            html_content = output_path.read_text(encoding="utf-8")

            # 3. Publish to WordPress
            slug = clinic_name.lower().replace(" ", "-").replace(".", "-")
            async with httpx.AsyncClient(timeout=60.0) as client:
                response = await client.post(
                    f"{WP_API_BASE}/wp-json/wp/v2/pages",
                    json={
                        "title": f"{clinic_name} — Пресейл-отчёт AIM",
                        "content": html_content,
                        "slug": f"report-{slug}-{session_hash[:6]}",
                        "status": "publish",  # Public immediately
                        "meta": {
                            "session_hash": session_hash,
                            "generated_at": str(Path(output_path).stat().st_mtime)
                        }
                    },
                    # TODO: Add WordPress auth (application password or JWT)
                    # auth=("aim-bot", WP_AUTH)
                )
                response.raise_for_status()
                wp_data = response.json()
                report_url = wp_data.get("link", "")

                logger.info("Report published: %s", report_url)

                # 4. Push SSE event (report-ready per D-19)
                from app.main import push_tool_progress
                push_tool_progress(
                    stage="report",
                    message=f"Отчёт готов: {report_url}"
                )

                return json.dumps({
                    "report_url": report_url,
                    "session_hash": session_hash,
                    "status": "published"
                }, ensure_ascii=False)

        except httpx.HTTPStatusError as e:
            logger.error("WordPress API error: %s", e)
            return json.dumps({
                "error": "Failed to publish to WordPress",
                "status": e.response.status_code,
                "detail": str(e)
            })
        except Exception as e:
            logger.exception("Report generation failed")
            return json.dumps({"error": str(e)})


    registry.register(
        name="generate_report",
        toolset="aim-operations",
        schema={
            "name": "generate_report",
            "description": "Generate HTML presale report and publish to WordPress",
            "parameters": {
                "type": "object",
                "properties": {
                    "session_hash": {"type": "string", "description": "8-char session ID"},
                    "clinic_name": {"type": "string", "description": "Clinic name"},
                    "orchestrator_data": {"type": "object", "description": "Full orchestrator output"}
                },
                "required": ["session_hash", "clinic_name", "orchestrator_data"]
            }
        },
        handler=handle_generate_report,
        check_fn=lambda: True,
        is_async=True,
        description="Generate and publish presale report to WordPress",
        emoji="📄"
    )
    ```
