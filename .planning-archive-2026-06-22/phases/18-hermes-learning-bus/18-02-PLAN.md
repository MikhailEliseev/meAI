### Task 2.1: Teacher → Hermes Knowledge Sync

**Файл создать:** `AIM/hermes/knowledge/teacher_sync.py`

**Что делает:**
```python
class TeacherSync:
    """Синхронизация знаний Teacher → Hermes"""
    
    def __init__(self, teacher: TeacherAgent, vault: HermesKnowledgeVault):
        self.teacher = teacher
        self.vault = vault
    
    async def sync_domain(self, domain: str):
        """Синхронизировать знания по домену из Qdrant в Hermes vault"""
        # 1. Запросить топ-знания из Qdrant
        results = await self.teacher.search_knowledge(
            query=f"best practices {domain}",
            domain=f"{domain}_knowledge",
            limit=10
        )
        
        # 2. Сохранить в wiki/learnings/{domain}/
        for result in results:
            await self.vault.store_learning(domain, {
                "source": "teacher",
                "content": result.content,
                "quality_score": result.score,
                "stored_at": datetime.now().isoformat()
            })
    
    async def sync_all_domains(self):
        """Синхронизировать все домены"""
        for domain in ["seo", "content", "ads", "general"]:
            await self.sync_domain(domain)
```

### Task 2.2: Magisters → Hermes Context Query

**Файл создать:** `AIM/src/aim/integration/hermes_context.py`

**Что делает:**
```python
class HermesContextProvider:
    """Провайдер контекста от Hermes для Magisters"""
    
    def __init__(self, hermes_url: str = "http://localhost:8000"):
        self.hermes_url = hermes_url
    
    async def get_context(self, domain: str, action: str) -> dict:
        """Получить контекст от Hermes перед делегированием"""
        async with httpx.AsyncClient(timeout=5.0) as client:
            resp = await client.get(
                f"{self.hermes_url}/api/knowledge/context",
                params={"domain": domain, "action": action}
            )
            resp.raise_for_status()
            return resp.json()
    
    async def enrich_task(self, domain: str, task: Task) -> Task:
        """Обогатить задачу контекстом от Hermes"""
        context = await self.get_context(domain, task.action)
        task.payload["hermes_context"] = context
        task.payload["enriched"] = True
        return task
```
