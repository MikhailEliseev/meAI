### C1: Специализация — dominance-based вместо first-match-wins

Функция `_detect_specialization()` (строка 282) сейчас итерирует `_SPECIALIZATIONS` в фиксированном порядке и возвращает первую специализацию, где хоть один keyword совпал.

**Заменить на:** подсчитать количество keyword-совпадений для КАЖДОЙ специализации, выбрать ту, у которой максимум совпадений.

Конкретный алгоритм:
```python
def _detect_specialization(text_lower: str, url: str) -> str:
    # Считаем keyword-совпадения для каждой специализации
    match_counts: dict[str, int] = {}
    for spec, patterns in _SPECIALIZATIONS.items():
        count = sum(1 for p in patterns if p in text_lower)
        if count > 0:
            match_counts[spec] = count
    
    # Выбираем специализацию с максимальным числом совпадений
    if match_counts:
        return max(match_counts, key=match_counts.get)
    
    # Fallback: URL (без изменений)
    url_lower = url.lower()
    if "stomat" in url_lower or "dent" in url_lower:
        return "стоматология"
    if "cosmet" in url_lower or "kosmet" in url_lower:
        return "косметология"
    if "clinic" in url_lower or "med" in url_lower or "клиник" in url_lower:
        return "многопрофильная клиника"
    return ""
```

**Важно:** При ничьей (равное число совпадений) — сохранить порядок итерации dict (Python 3.7+ гарантирует insertion order), что отдаст приоритет первой в `_SPECIALIZATIONS`. Это консервативное поведение.

### C3: Фильтрация отрицательного контекста в детекции услуг

Функция `_detect_services()` (строка 271) сейчас чисто substring-матчит — любое вхождение keyword = услуга обнаружена.

**Добавить негативный фильтр.** Для каждого найденного keyword проверить, что окружающий контекст НЕ содержит маркеры отрицания или противопоказаний.

Алгоритм:
1. Для каждого сервиса из `_MEDICAL_SERVICES` ищем ВСЕ вхождения всех его паттернов в тексте.
2. Для каждого найденного вхождения извлекаем контекст: 80 символов ДО и 40 символов ПОСЛЕ позиции совпадения.
3. Проверяем контекст на негативные маркеры. Если хотя бы один маркер найден — это вхождение игнорируем.
4. Если осталось хотя бы одно позитивное вхождение — сервис считается обнаруженным.

Негативные маркеры (регистронезависимый поиск в контексте):
```python
_NEGATION_MARKERS: list[str] = [
    "противопоказани",   # противопоказания к имплантации
    "не используем",      # не используем ботокс
    "не применяем",       # не применяем имплантацию
    "не проводим",        # не проводим хирургию
    "не делаем",          # не делаем отбеливание
    "отказались от",      # отказались от ботокса
    "нельзя",             # нельзя делать имплантацию
    "запрещен",           # запрещено
    "не рекомендуется",   # не рекомендуется
    "не показан",         # не показана имплантация
    "не является",        # не является услугой
]
```

**Важно:** Проверка ТОЛЬКО в пределах контекстного окна (±80/±40 символов от вхождения), не глобально. Отрицание про «ботокс» не должно блокировать «косметологию» в другом разделе страницы.

```python
def _detect_services(text_lower: str) -> list[str]:
    """Detect medical services from page text, excluding negation contexts."""
    _NEGATION_MARKERS: list[str] = [
        "противопоказани", "не используем", "не применяем", "не проводим",
        "не делаем", "отказались от", "нельзя", "запрещен",
        "не рекомендуется", "не показан", "не является",
    ]
    
    found: list[str] = []
    for service, patterns in _MEDICAL_SERVICES.items():
        service_detected = False
        for pattern in patterns:
            idx = text_lower.find(pattern)
            while idx != -1:
                # Извлекаем контекст
                ctx_start = max(0, idx - 80)
                ctx_end = min(len(text_lower), idx + len(pattern) + 40)
                context = text_lower[ctx_start:ctx_end]
                
                # Проверяем негативные маркеры
                negated = any(marker in context for marker in _NEGATION_MARKERS)
                if not negated:
                    service_detected = True
                    break  # достаточно одного позитивного вхождения
                
                # Ищем следующее вхождение
                idx = text_lower.find(pattern, idx + 1)
            
            if service_detected:
                found.append(service)
                break  # переходим к следующему сервису
    
    return found
```

```python
def _extract_city_from_schema(html: str) -> str:
    """Extract city from JSON-LD schema.org markup."""
    try:
        from bs4 import BeautifulSoup
        import json as _json
        
        soup = BeautifulSoup(html, "html.parser")
        for script in soup.find_all("script", type="application/ld+json"):
            try:
                data = _json.loads(script.string or "")
            except (_json.JSONDecodeError, TypeError):
                continue
            
            # Handle both single object and array
            items = data if isinstance(data, list) else [data]
            for item in items:
                if not isinstance(item, dict):
                    continue
                
                # Extract city from address
                address = item.get("address", {})
                if isinstance(address, dict):
                    locality = address.get("addressLocality", "")
                    if locality:
                        for city in _RUSSIAN_CITIES:
                            if _city_matches(locality, city):
                                return city
                
                # Also check location/area
                location = item.get("location", {})
                if isinstance(location, dict):
                    addr = location.get("address", {})
                    if isinstance(addr, dict):
                        locality = addr.get("addressLocality", "")
                        if locality:
                            for city in _RUSSIAN_CITIES:
                                if _city_matches(locality, city):
                                    return city
    except Exception:
        pass
    return ""
```

def _detect_city(text: str) -> str:
    """Detect city from page text (full page, no 5000 char limit)."""
    # Strategy 1: "в Городе" pattern
    city_preposition = re.search(
        r"\bв\s+(?:гор\.?\s*)?([А-ЯЁ][а-яё]+(?:[\s-][А-ЯЁ][а-яё]+)?)\b",
        text,  # ← БЫЛО text[:5000]

        if pattern.search(text):  # ← БЫЛО text_head = text[:5000]; pattern.search(text_head)

    return ""

### M1: Заменить TF-IDF на чистый Jaccard в _score_services()

Функция `_score_services()` (строка 1034) сейчас:
```python

```

**Заменить на:**
```python

```

TF-IDF на 5-10 словах не несёт полезного сигнала (это просто keyword overlap с лишней математикой). Чистый Jaccard прозрачен и корректен для коротких строк.

**Заодно упростить тело функции** — убрать вызов `_tfidf_cosine()` и построение `client_text`/`cand_text`:
```python
def _score_services(client: ClientProfile, candidate: CompanyProfile) -> float:
    """Score service overlap using Jaccard similarity.
    
    Pure Jaccard — competitor services are constructed (not scraped), 
    so TF-IDF on 5-10 word strings is misleading.
    """
    if not client.services:
        return 0.5  # neutral
    
    candidate_services = _candidate_services(client, candidate)
    if not candidate_services:
        return 0.1
    
    client_set = set(client.services)
    cand_set = set(candidate_services)
    jaccard = len(client_set & cand_set) / max(len(client_set | cand_set), 1)
    return round(jaccard, 4)
```

class TestScoreServicesJaccard:
    """M1: _score_services uses pure Jaccard."""

    def _make_client(self, services: list[str]) -> ClientProfile:
        return ClientProfile(
            url="https://test.ru",
            specialization="стоматология",
            city="Москва",
            services=services,
            estimated_revenue=30_000_000,

    def _make_candidate(self, source_spec: str = "стоматология") -> CompanyProfile:
        return CompanyProfile(
            inn="1234567890",
            legal_name="Тестовая Клиника",
            source_specialization=source_spec,
            okved_main="86.23",
            okved_secondary=[],
        )

    def test_full_overlap(self):
        """Полное пересечение → Jaccard = 1.0."""
        client = self._make_client(["терапия", "ортопедия"])
        # Стоматология candidate services include терапия, ортопедия, хирургия, etc.
        cand = self._make_candidate("стоматология")
        score = _score_services(client, cand)
        # client: {терапия, ортопедия}, candidate (стоматология): {стоматология, терапия, хирургия, ортопедия, ортодонтия, имплантация, гигиена}
        # Intersection: {терапия, ортопедия} = 2
        # Union: {стоматология, терапия, хирургия, ортопедия, ортодонтия, имплантация, гигиена} = 7
        # Jaccard = 2/7 ≈ 0.2857
        expected = 2 / 7
        assert abs(score - expected) < 0.01, f"Expected ~{expected:.4f}, got {score}"

    def test_no_overlap(self):
        """Нет пересечения → Jaccard = 0.0."""
        client = self._make_client(["пластическая хирургия"])
        cand = self._make_candidate("стоматология")
        score = _score_services(client, cand)
        assert score == 0.0, f"Expected 0.0, got {score}"

    def test_empty_client_services_neutral(self):
        """Нет услуг у клиента → 0.5 (neutral)."""
        client = self._make_client([])
