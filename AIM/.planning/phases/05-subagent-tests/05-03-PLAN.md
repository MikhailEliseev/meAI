@pytest.mark.asyncio

    # Mock original content
    original_content = "Dental implants cost between $1,000 and $3,000."
    
    # Mock OpenAI revision response

        "tokens_used": 200,
    }
    
    # Create task
    task = Task(
        task_id="test-writer-003",
        task_type="content_revision",
        parameters={
            "original_content": original_content,
            "feedback": "Add more detail about cost factors. Improve readability.",

    )
    
    # Execute

    improvements = result.data["improvements"]
    assert len(improvements) > 0
    
    # Verify LLM call with feedback
