"""Unit tests for Analytics Agent

Tests metrics collection, data validation, and report generation.

import pytest
from unittest.mock import AsyncMock

from meai.agents.base_agent import Task
from src.aim.subagents.analytics_agent import AnalyticsAgent
from tests.fixtures.subagent_data import GA4_METRICS, YANDEX_METRICS

@pytest.mark.asyncio
async def test_metrics_collection_success(analytics_agent, mock_api_clients):
    """Test metrics collection from GA4 and Yandex Metrica"""
    # Mock GA4 API response
    mock_api_clients["ga4"].get_metrics.return_value = GA4_METRICS
    
    # Mock Yandex Metrica API response
    mock_api_clients["yandex_metrica"].get_metrics.return_value = YANDEX_METRICS
    
    # Create task
    task = Task(
        task_id="test-analytics-001",
        task_type="metrics_collection",
        parameters={
            "date_range": "last_30_days",
            "metrics": ["sessions", "users", "bounce_rate", "conversions"],
            "sources": ["ga4", "yandex"],
        },
    )
    
    # Execute
    result = await analytics_agent.execute_task(task)
    
    # Verify success
    assert result.status == "success"
    assert "metrics" in result.data
    
    # Verify metrics collected
    metrics = result.data["metrics"]
    assert "ga4" in metrics
    assert "yandex" in metrics
    
    # Verify GA4 metrics
    ga4 = metrics["ga4"]
    assert ga4["sessions"] == 15000
    assert ga4["users"] == 12000
    assert ga4["bounce_rate"] == 0.45
    assert ga4["conversions"] == 150
    
    # Verify Yandex metrics
    yandex = metrics["yandex"]
    assert yandex["visits"] == 14000
    assert yandex["visitors"] == 11000
    assert yandex["bounce_rate"] == 0.48
    assert yandex["goals"] == 140
    
    # Verify API calls
    mock_api_clients["ga4"].get_metrics.assert_called_once()
    mock_api_clients["yandex_metrica"].get_metrics.assert_called_once()
    
    # Verify data aggregation
    assert "aggregated" in result.data
    aggregated = result.data["aggregated"]
    assert aggregated["total_sessions"] == 15000 + 14000
    assert aggregated["total_conversions"] == 150 + 140
