"""Unit tests for Keyword Research Agent

Tests business logic, API integration, compliance, and priority calculation.
"""

import pytest
from unittest.mock import AsyncMock

from meai.agents.base_agent import Task
from src.aim.subagents.keyword_research_agent import KeywordResearchAgent
from tests.fixtures.subagent_data import MEDICAL_KEYWORDS, SEMRUSH_RESPONSE, AHREFS_RESPONSE
from tests.fixtures.subagent_fixtures import mock_api_clients, keyword_research_agent


@pytest.mark.asyncio
async def test_keyword_expansion_success(keyword_research_agent, mock_api_clients):
    """Test successful keyword expansion through SEMrush"""
    # Mock SEMrush response
    mock_api_clients["semrush"].expand_keywords.return_value = SEMRUSH_RESPONSE["keywords"]
    
    # Create task
    task = Task(
        task_id="test-001",
        task_type="keyword_research",
        parameters={
            "seed_keyword": "dental implants",
            "max_keywords": 100,
            "min_volume": 1000,
        },
    )
    
    # Execute
    result = await keyword_research_agent.execute_task(task)
    
    # Verify success
    assert result.status == "success"
    assert len(result.data["keywords"]) == 2
    assert result.data["source"] == "semrush"
    
    # Verify API call
    mock_api_clients["semrush"].expand_keywords.assert_called_once()
    call_args = mock_api_clients["semrush"].expand_keywords.call_args
    assert call_args[1]["seed_keyword"] == "dental implants"
    assert call_args[1]["max_keywords"] == 100
    
    # Verify cost tracking
    assert keyword_research_agent.total_cost_usd > 0
    assert keyword_research_agent.api_calls == 1


@pytest.mark.asyncio
async def test_keyword_expansion_with_fallback(keyword_research_agent, mock_api_clients):
    """Test fallback to Ahrefs when SEMrush fails (circuit breaker)"""
    # Mock SEMrush failures (circuit breaker opens after 5 failures)
    mock_api_clients["semrush"].expand_keywords.side_effect = [
        ConnectionError("API timeout"),
        ConnectionError("API timeout"),
        ConnectionError("API timeout"),
        ConnectionError("API timeout"),
        ConnectionError("API timeout"),
    ]
    
    # Mock Ahrefs success
    mock_api_clients["ahrefs"].expand_keywords.return_value = AHREFS_RESPONSE["keywords"]
    
    # Create task
    task = Task(
        task_id="test-002",
        task_type="keyword_research",
        parameters={
            "seed_keyword": "dental implants",
            "max_keywords": 100,
        },
    )
    
    # Execute
    result = await keyword_research_agent.execute_task(task)
    
    # Verify fallback worked
    assert result.status == "success"
    assert result.data["source"] == "ahrefs"
    assert len(result.data["keywords"]) == 1
    
    # Verify SEMrush circuit breaker opened
    assert mock_api_clients["semrush"].expand_keywords.call_count == 5
    
    # Verify Ahrefs fallback called
    mock_api_clients["ahrefs"].expand_keywords.assert_called_once()


@pytest.mark.asyncio
async def test_compliance_blocking(keyword_research_agent, mock_api_clients):
    """Test blocking of risky keywords (FDA/HIPAA compliance)"""
    # Mock SEMrush response with risky keyword
    mock_api_clients["semrush"].expand_keywords.return_value = [
        {
            "keyword": "buy oxycodone online",
            "search_volume": 5000,
            "keyword_difficulty": 45,
            "cpc": 12.00,
            "intent": "transactional",
        },
    ]
    
    # Create task
    task = Task(
        task_id="test-003",
        task_type="keyword_research",
        parameters={
            "seed_keyword": "buy oxycodone online",
            "max_keywords": 100,
        },
    )
    
    # Execute
    result = await keyword_research_agent.execute_task(task)
    
    # Verify blocking
    assert result.status == "blocked"
    assert "compliance" in result.error.lower()
    assert result.data["risk_level"] == "high"
    assert "buy oxycodone online" in result.data["blocked_keywords"]
    
    # Verify audit trail saved
    assert result.data["audit_trail_id"] is not None
