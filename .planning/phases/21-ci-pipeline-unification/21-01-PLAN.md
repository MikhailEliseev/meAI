```python
class Agent(ABC):
    def __init__(self, agent_id, agent_type, database_url, vault_path="./obsidian", event_bus=None):
        self.event_bus = event_bus if event_bus is not None else EventBus(database_url)

    async def initialize(self):
        await self.db.connect()
        await self.vault.initialize()
        await self.event_bus.initialize()
        await self._create_tables()
        self._listener_task = asyncio.create_task(self._listen_for_tasks())

    async def _listen_for_tasks(self):
        """Polls every 2s for pending task.request Messages targeting this agent."""
        while True:
            messages = await self.event_bus.get_messages(
                agent_id=self.agent_id, status="pending", limit=10
            )
            for msg in messages:
                if msg.message_type != "task.request":
                    continue
                task_data = msg.payload.get("task", msg.payload)
                task = Task(**task_data)
                await self.receive_task(task)
                await self.event_bus.mark_processed(msg.message_id)
            await asyncio.sleep(2)

    async def receive_task(self, task):
        task.status = TaskStatus.RECEIVED
        await self._store_task(task)
        await self._write_task_to_vault(task)
        self.active_tasks[task.subtask_id] = task
        await self._execute_and_report(task)

    async def _execute_and_report(self, task):
        result = await self.execute_task(task)  # abstract, implemented by each CI agent
        await self.report_result(result)

    async def report_result(self, result):
        # Publishes agent.result Message with to_agent="operator"
        message = Message(from_agent=self.agent_id, to_agent="operator",
                          message_type="agent.result", ...)
        await self.event_bus.publish(message)
```

From meai/events/event_bus.py:
```python
class EventBus:
    _subscribers: dict[str, list[Callable]]  # event_type -> handlers

    def subscribe(self, event_type: str, handler: Callable[[Event], Awaitable]):
        """Register handler for event_type. Handler fires on publish()."""

    async def publish(self, event: Event | Message):
        """If Event: notifies subscribers + stores. If Message: stores only."""

    async def get_messages(self, agent_id, status="pending", limit=100):
        """Get Messages where to_agent == agent_id."""
```

```python
# Bridge: publish ci.agent.completed Event when agent reports result
_original_report = agent.report_result

async def _bridged_report(result):
    # Call original (publishes agent.result Message to operator)
    await _original_report(result)
    # Also publish ci.agent.completed Event for orchestrator's callback
    # The correlation_id was passed through task.data by _execute_single_phase
    # We need to extract it from the result's subtask_id
    try:
        # Find active task to get correlation_id from task.data
        for t in agent.active_tasks.values():
            if t.subtask_id == result.subtask_id:
                correlation_id = t.data.get("correlation_id", "")
                break
        else:
            correlation_id = ""
    except Exception:
        correlation_id = ""
    
    await self.event_bus.publish(Event(
        event_type="ci.agent.completed",
        payload={
            "correlation_id": correlation_id,
            "agent": agent_name,
            "phase": int(result.subtask_id.split("-")[-1]) if "-" in result.subtask_id else 0,
            "status": result.status,
            "result": result.result if isinstance(result.result, dict) else {},
        }
    ))

agent.report_result = _bridged_report
```
