```python
"""Apify async client with resilience patterns — circuit breaker, retry, rate limiting.

Used by all Apify actor integrations: Google Maps, Website Crawler, Instagram.
"""

import os
import logging
from datetime import timedelta
from typing import Optional

from apify_client import ApifyClientAsync

logger = logging.getLogger(__name__)

# ── Resilience ────────────────────────────────────────────────────────
# Circuit breaker: opens after 5 consecutive failures, resets after 60s
# Retry: exponential backoff 1s → 2s → 4s → 8s → 16s → 30s (max)
# Token bucket: 10 requests per second burst, refill 2/sec sustained

_CIRCUIT_BREAKER_MAX_FAILURES = 5
_CIRCUIT_BREAKER_RESET_SECONDS = 60
_RETRY_MAX_ATTEMPTS = 3
_RETRY_BASE_DELAY = 1.0
_RETRY_MAX_DELAY = 30.0
_RATE_LIMIT_CAPACITY = 10
_RATE_LIMIT_REFILL_RATE = 2.0

# ── Actor pricing (USD per 1000 results) ──────────────────────────────
_ACTOR_PRICING = {
    "compass/crawler-google-places": 5.0,       # $5/1000 results
    "apify/website-content-crawler": 10.0,       # $10/1000 pages
    "apify/instagram-scraper": 5.0,              # $5/1000 results
}


class ApifyClient:
    """Shared Apify client with resilience patterns.

    Usage:
        client = get_apify_client()
        results = await client.run_actor(
            "compass/crawler-google-places",
            run_input={"searchString": "стоматология", "location": "Москва"},
            max_results=50,
        )
    """

    def __init__(self, token: Optional[str] = None):
        self.token = token or os.getenv("APIFY_API_TOKEN", "")
        if not self.token:
            logger.warning("APIFY_API_TOKEN not set — ApifyClient will be a no-op")
        self._client: Optional[ApifyClientAsync] = None
        self._failure_count = 0
        self._last_failure_time: float = 0.0
        self._circuit_open = False

    @property
    def configured(self) -> bool:
        return bool(self.token)

    def _get_client(self) -> ApifyClientAsync:
        if self._client is None and self.token:
            self._client = ApifyClientAsync(token=self.token)
        return self._client

    # ── Circuit breaker ──────────────────────────────────────────────

    def _check_circuit(self):
        """Raise if circuit breaker is open."""
        if self._circuit_open:
            import time
            elapsed = time.monotonic() - self._last_failure_time
            if elapsed > _CIRCUIT_BREAKER_RESET_SECONDS:
                logger.info("ApifyClient: circuit breaker reset (%.0fs elapsed)", elapsed)
                self._circuit_open = False
                self._failure_count = 0
            else:
                raise RuntimeError(
                    f"ApifyClient: circuit breaker open — {_CIRCUIT_BREAKER_RESET_SECONDS - elapsed:.0f}s until reset"
                )

    def _record_success(self):
        self._failure_count = 0
        self._circuit_open = False

    def _record_failure(self, error: str):
        import time
        self._failure_count += 1
        self._last_failure_time = time.monotonic()
        if self._failure_count >= _CIRCUIT_BREAKER_MAX_FAILURES:
            self._circuit_open = True
            logger.error(
                "ApifyClient: circuit breaker OPEN after %d failures — %s",
                self._failure_count, error,
            )

    # ── Main API ─────────────────────────────────────────────────────

    async def run_actor(
        self,
        actor_id: str,
        run_input: dict,
        run_timeout: timedelta = timedelta(minutes=5),
        memory_mbytes: int = 2048,
        max_results: int = 50,
    ) -> list[dict]:
        """Run an Apify actor and return its dataset items.

        Args:
            actor_id: Actor ID (e.g. "compass/crawler-google-places")
            run_input: Actor-specific input parameters
            run_timeout: Maximum run duration
            memory_mbytes: Memory allocation for the actor
            max_results: Maximum number of results to return

        Returns:
            List of result items from the actor's default dataset.
            Returns empty list on any error (graceful degradation).
        """
        self._check_circuit()

        if not self.configured:
            logger.warning("ApifyClient: skipping run_actor(%s) — not configured", actor_id)
            return []

        client = self._get_client()
        if client is None:
            return []

        for attempt in range(_RETRY_MAX_ATTEMPTS):
            try:
                logger.info(
                    "ApifyClient: starting actor %s (attempt %d/%d)",
                    actor_id, attempt + 1, _RETRY_MAX_ATTEMPTS,
                )
                run = await client.actor(actor_id).call(
                    run_input=run_input,
                    run_timeout=run_timeout,
                    memory_mbytes=memory_mbytes,
                )

                if run is None or not run.get("defaultDatasetId"):
                    logger.warning("ApifyClient: actor %s returned no dataset", actor_id)
                    self._record_success()
                    return []

                dataset = client.dataset(run["defaultDatasetId"])
                results = []
                async for item in dataset.iterate_items():
                    results.append(item)
                    if len(results) >= max_results:
                        break

                self._record_success()
                logger.info(
                    "ApifyClient: actor %s completed — %d results",
                    actor_id, len(results),
                )
                return results

            except Exception as e:
                error_msg = f"{type(e).__name__}: {e}"
                logger.warning(
                    "ApifyClient: actor %s failed (attempt %d/%d) — %s",
                    actor_id, attempt + 1, _RETRY_MAX_ATTEMPTS, error_msg,
                )
                if attempt < _RETRY_MAX_ATTEMPTS - 1:
                    delay = min(_RETRY_BASE_DELAY * (2 ** attempt), _RETRY_MAX_DELAY)
                    import asyncio
                    await asyncio.sleep(delay)
                else:
                    self._record_failure(error_msg)

        return []

    async def get_results(
        self,
        dataset_id: str,
        max_results: int = 100,
    ) -> list[dict]:
        """Get results from an existing dataset by ID."""
        if not self.configured:
            return []

        client = self._get_client()
        if client is None:
            return []

        try:
            dataset = client.dataset(dataset_id)
            results = []
            async for item in dataset.iterate_items():
                results.append(item)
                if len(results) >= max_results:
                    break
            return results
        except Exception as e:
            logger.error("ApifyClient: get_results failed — %s", e)
            return []

    def estimate_cost(
        self,
        actor_id: str,
        result_count: int = 50,
    ) -> float:
        """Estimate cost for an actor run in USD.

        Args:
            actor_id: Actor ID
            result_count: Expected number of results

        Returns:
            Estimated cost in USD.
        """
        price_per_1k = _ACTOR_PRICING.get(actor_id, 10.0)
        return (result_count / 1000.0) * price_per_1k


# ── Singleton ───────────────────────────────────────────────────────────

_apify_client: ApifyClient | None = None


def get_apify_client() -> ApifyClient:
    """Get or create the shared ApifyClient singleton."""
    global _apify_client
    if _apify_client is None:
        _apify_client = ApifyClient()
    return _apify_client
```

# Categories that indicate a medical organization vs unrelated business
_MEDICAL_CATEGORIES = {

_STATE_NAME_PATTERNS = [

]


def _is_state_healthcare(name: str) -> bool:
    """Check if organization name indicates a state healthcare institution."""
    if not name:
        return False
    name_lower = name.lower()
    return any(pattern in name_lower for pattern in _STATE_NAME_PATTERNS)


def _is_medical_category(categories: list[dict]) -> bool:
    """Check if any Google Maps category indicates a medical organization."""
    if not categories:
        return True  # no categories → keep (don't filter out)
    for cat in categories:
        cat_name = (cat.get("name") or "").lower()
        if cat_name in _MEDICAL_CATEGORIES:
            return True
        # Partial match for compound categories
        for med_cat in _MEDICAL_CATEGORIES:
            if med_cat in cat_name:
                return True
    return False


async def find_medical_orgs(
    specialization: str,
    city: str,
    count: int = 50,
) -> list[CompanyProfile]:
    """Find medical organizations via Apify Google Maps Scraper.

    Args:
        specialization: Medical specialization (e.g. "стоматология")
        city: City name (e.g. "Москва")
        count: Maximum number of results (default 50)

    Returns:
        List of CompanyProfile objects with digital presence fields populated.
        Returns empty list on any error (graceful degradation).
    """

    if not client.configured:
        logger.warning("Apify Google Maps: not configured — skipping")
        return []

    query = f"{specialization} {city}"

    results = await client.run_actor(
        actor_id="compass/crawler-google-places",
        run_input={
            "searchString": query,
            "location": city,
            "maxResults": count,
            "language": "ru",
            "includeSocialMedia": True,
            "includeReviews": False,  # дороже, не нужно для discovery
            "proxyConfig": {"useApifyProxy": True},
        },
        max_results=count,
    )

    if not results:
        logger.info("Apify Google Maps: 0 results for '%s' in %s", specialization, city)
        return []

    profiles: list[CompanyProfile] = []
    for item in results:
        # Skip permanently closed
        if item.get("permanentlyClosed"):
            continue

        name = item.get("title", "").strip()
        if not name:
            continue

        # Skip state healthcare
        if _is_state_healthcare(name):
            logger.debug("Apify Google Maps: skipping state org '%s'", name[:50])
            continue

        # Skip non-medical categories (but keep if no categories — could be unlabeled medical)
        categories = item.get("categories", [])
        if categories and not _is_medical_category(categories):
            logger.debug("Apify Google Maps: skipping non-medical '%s' — %s", name[:50], categories)
            continue

        # Extract social media from Google Maps card
        social_media = item.get("socialMedia", {}) or {}

        # Build CompanyProfile
        profile = CompanyProfile(
            inn="",  # no INN from Google Maps — DaData enrichment fills this
            legal_name=name,
            website=item.get("website"),
            social_links=social_media,
            geo_lat=_safe_get(item, "location", "lat"),
            geo_lon=_safe_get(item, "location", "lng"),
            rating=item.get("rating"),
            reviews_count=item.get("reviewsCount"),
            legal_address=item.get("address"),
            source_specialization=specialization,
            data_source="apify_google_maps",
            confidence=0.65,  # Google Maps is reliable but not 100%
        )
        profiles.append(profile)

    logger.info(
        "Apify Google Maps: found %d medical orgs for '%s' in %s (out of %d raw results)",
        len(profiles), specialization, city, len(results),
    )
    return profiles


def _safe_get(d: dict, *keys) -> Optional[float]:
    """Safely extract nested dict value."""
    for key in keys:
        if isinstance(d, dict):
            d = d.get(key)
        else:
            return None
    return d if isinstance(d, (int, float)) else None

"""Apify Instagram Scraper — social signals for competitor enrichment.

Lightweight, non-blocking. Not all medical clinics have Instagram.
Actor: apify/instagram-scraper
Cost: ~$0.01 per 2-3 profiles
"""

import logging
import re
from typing import Optional
from urllib.parse import urlparse

from .apify_client import get_apify_client

logger = logging.getLogger(__name__)


def _extract_username(website: Optional[str]) -> Optional[str]:
    """Extract Instagram username from website URL or company name.

    Tries common patterns: instagram.com/username, website subdomain, etc.
    """
    if not website:
        return None

    # Direct Instagram URL
    if "instagram.com" in website:
        path = urlparse(website).path.strip("/")
        return path.split("/")[0] if path else None

    # Clean domain name (e.g. clinic-name.ru → clinic_name)
    domain = website
    for prefix in ("https://", "http://", "www."):
        domain = domain.removeprefix(prefix)
    domain = domain.split("/")[0].split(".")[0]
    # Sanitize: only letters, numbers, underscores, dots
    domain = re.sub(r"[^a-zA-Z0-9_.]", "", domain)
    return domain if len(domain) >= 3 else None


async def discover_social_presence(
    company_name: str,
    website: Optional[str] = None,
) -> dict:
    """Discover Instagram presence for a competitor clinic.

    Args:
        company_name: Clinic name (e.g. "Стоматология Премиум")
        website: Clinic website URL (used to guess Instagram username)

    Returns:
        Dict with keys: instagram_url, followers, bio, posts_count.
        Empty dict if no Instagram profile found or on error.
    """
    client = get_apify_client()
    if not client.configured:
        logger.warning("Apify Instagram: not configured — skipping")
        return {}

    # Try: website-derived username first, then company name
    candidates: list[str] = []
    username = _extract_username(website)
    if username:
        candidates.append(username)

    # Clean company name as fallback
    clean_name = re.sub(r"[^a-zA-Zа-яА-Я0-9_]", "", company_name.lower())
    clean_name = re.sub(r"[а-яё]+", lambda m: _transliterate(m.group()), clean_name)
    clean_name = re.sub(r"[^a-z0-9_.]", "", clean_name)
    if clean_name and clean_name not in candidates:
        candidates.append(clean_name[:30])

    if not candidates:
        return {}

    results = await client.run_actor(
        actor_id="apify/instagram-scraper",
        run_input={
            "usernames": candidates,
            "resultsLimit": 5,
        },
        max_results=5,
        run_timeout=__import__("datetime").timedelta(minutes=2),
        memory_mbytes=512,
    )

    if not results:
        return {}

    # Take the first matching profile
    for item in results:
        username_found = item.get("username", "")
        followers = item.get("followersCount", 0) or item.get("followers", 0)
        bio = item.get("biography", "") or item.get("bio", "")

        if username_found:
            logger.info(
                "Apify Instagram: found @%s — %d followers for '%s'",
                username_found, followers, company_name[:30],
            )

                "instagram_url": f"https://instagram.com/{username_found}",
                "followers": followers,
                "bio": bio,
                "posts_count": item.get("postsCount", 0),
            }

    return {}


def _transliterate(text: str) -> str:
    """Rough Cyrillic → Latin transliteration for username guessing."""
    _MAP = {
        "а": "a", "б": "b", "в": "v", "г": "g", "д": "d", "е": "e", "ё": "e",
        "ж": "zh", "з": "z", "и": "i", "й": "y", "к": "k", "л": "l", "м": "m",

        "ф": "f", "х": "h", "ц": "ts", "ч": "ch", "ш": "sh", "щ": "sch",
        "ъ": "", "ы": "y", "ь": "", "э": "e", "ю": "yu", "я": "ya",
    }
    result = ""
    for ch in text:
        result += _MAP.get(ch, ch)
    return result

```python
async def _enrich_with_web_crawler(
    self,
    candidates: list[CompanyProfile],
    client: ClientProfile,
) -> None:
    """Extract real medical services from competitor websites.
    
    Runs Website Content Crawler for each candidate that has a website.
    Updates CompanyProfile.scraped_services in-place.
    """
    with_website = [c for c in candidates if c.website]
    if not with_website:
        return
    
    async def _crawl_one(c: CompanyProfile):
        try:
            services = await extract_services_from_website(c.website)
            if services:
                c.scraped_services = services
                logger.info(
                    "WebCrawler: %s → %d real services: %s",
                    c.legal_name[:30], len(services), services[:5],
                )
        except Exception as e:
            logger.debug("WebCrawler failed for %s: %s", c.website, e)
    
    # Crawl up to 5 websites in parallel
    tasks = [_crawl_one(c) for c in with_website[:5]]
    await asyncio.gather(*tasks, return_exceptions=True)
```
